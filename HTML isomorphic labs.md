    const initAccordion = () => {

        const accordionKey = 'accordion-element'
        const accordionRootAttribute = 'root'
        const accordionTriggerAttribute = 'trigger'
        const accordionContentAttribute = 'content'
        const activeClass = 'is-open';

        const items = gsap.utils.toArray(`[${accordionKey}="${accordionRootAttribute}"]`);

        items.forEach(item => {
            const trigger = item.querySelector(`[${accordionKey}="${accordionTriggerAttribute}"]`);
            const content = item.querySelector(`[${accordionKey}="${accordionContentAttribute}"]`);
            const plus = item.querySelector(`.work-accordion_plus`)
            const minus = item.querySelector(`.work-accordion_minus`)
            gsap.set(content, {
            	opacity: 0,
              y: 5
            })
            content.querySelectorAll('a').forEach(a => a.setAttribute('tabindex', '-1'))
            trigger.addEventListener('click', () => {
                const isOpen = trigger.classList.contains(activeClass);
                if(isOpen) {
                    trigger.setAttribute('aria-expanded', 'false');
                    content.setAttribute('aria-hidden', 'true');
                    content.querySelectorAll('a').forEach(a => a.setAttribute('tabindex', '-1'))
                    trigger.classList.remove(activeClass);
                    gsap.to(content, {
                        height: 0,
                        duration: 0.3,
                        opacity: 0,
                        y: 5
                    });
                    plus.style.display = "block"
                    minus.style.display = "none"
                } else {
                    trigger.setAttribute('aria-expanded', 'true');
                    content.setAttribute('aria-hidden', 'false');
                    content.querySelectorAll('a').forEach(a => a.setAttribute('tabindex', '0'))
                    trigger.classList.add(activeClass);
                    gsap.to(content, {
                        height: 'auto',
                        opacity: 1,
                        duration: 0.3,
                        y: 0
                    });
                    plus.style.display = "none"
                    minus.style.display = "block"
                }
            });
        });
    }
